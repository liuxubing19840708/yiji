## 十七、数据模型总览（建议 ER）

```
user(id, phone, name, avatar, gender, birthday, password_hash, created_at)
company(id, name, city, company_no UNIQUE, capacity, created_at)
user_company(id, user_id, company_id, post_id, status[0待审/1在职/2离职], is_pinned, joined_at, hired_date, is_suspended, fixed_salary, fixed_type[day/month])
post(id, company_id, name, permissions JSON, is_system)
style(id, company_id, style_no, name, price DECIMAL(10,3), customer_id, image, is_deleted)
process(id, company_id, name, default_price, is_fixed, sort)
style_process(style_id, process_id, price)
production(id, company_id, style_id, bed_no, total_qty, order_type, customer_id, is_locked, is_cut_done, split_mode, fabric)
ticket(id, company_id, production_id, bed_no, bundle_no, color, size, qty, fabric, piece_no, qr_code, is_printed)   UNIQUE(bed_no, bundle_no)
record(id, company_id, ticket_id, process_id, user_id, operator_id, qty, price, registered_at, settlement_id NULL)
salary_adjust(id, company_id, user_id, type[deduct/reward], amount, reason, operator_id, created_at)
settlement(id, company_id, user_id NULL(公司级)/员工级, amount, period_start, period_end, employee_count, status, created_by, created_at)
settle_apply(id, company_id, user_id, amount, note, status[待审/已审/已发放/已作废/已拒绝], reviewed_by, paid_by, created_at)
audit_log(id, company_id, type, bed_no, style_no, bundle_no, target_user_id, operator_id, before JSON, after JSON, message, created_at)
notification(id, company_id, user_id, type, title, body, is_read, created_at)
customer(id, company_id, name, customer_no, group_id)
customer_group(id, company_id, name)
company_setting(company_id, key, value)
```

核心聚合查询（工资）：
```sql
SELECT r.user_id, SUM(r.qty * r.price) unsettled
FROM record r WHERE r.company_id=? AND r.settlement_id IS NULL
GROUP BY r.user_id;
```
